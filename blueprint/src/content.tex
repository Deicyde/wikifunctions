\chapter*{How to read this blueprint}

This blueprint maps the Lean~4 formalization in the \href{https://github.com/Deicyde/wikifunctions}{\texttt{Deicyde/wikifunctions}} repository onto the deployed WikiLambda / Wikifunctions system it models. Every node carries two kinds of link:

\begin{itemize}
  \item a \textbf{Lean} link (\leansrc{Wikifunctions/Core.lean}{43}-style) to the exact source line, and the declaration name shown by the \textsf{L$\exists\forall$N} badge;
  \item a \textbf{WikiLambda} link (e.g.\ \wf{Z13701}) to the live object on \href{https://www.wikifunctions.org}{wikifunctions.org} that the Lean models.
\end{itemize}

In the dependency graph, a node is filled green when its Lean statement \emph{and} proof are complete; edges are \texttt{\textbackslash uses} dependencies. Everything here is proved with no \texttt{sorry}; the two headline theorems depend only on \texttt{[propext, Classical.choice, Quot.sound]}.

The system is specified by a four-layer stack, cited throughout: the prose \href{https://www.wikifunctions.org/wiki/Wikifunctions:Function_model}{Function model}, the machine-enforced \href{https://gitlab.wikimedia.org/repos/abstract-wiki/wikifunctions/function-schemata}{function-schemata}, the live objects (via the \texttt{wikilambda\_fetch} API), and the deployed orchestrator/evaluator. The companion \href{https://github.com/Deicyde/wikifunctions/blob/main/BLUEPRINT.md}{BLUEPRINT.md} pins each mapping to a runnable check; \href{https://github.com/Deicyde/wikifunctions/blob/main/SPEC_AUDIT.md}{SPEC\_AUDIT.md} records the confirmed divergences (D1--D18).

\chapter{The object model and evaluator core}

\texttt{Core.lean} is an open, registry-parameterized model of the Wikifunctions composition language: the \wf{Z7} function-call / \wf{Z18} argument-reference fragment, evaluated against a wiki of \wf{Z8} functions.

\begin{definition}[ZID]\label{def:core-zid}
  \lean{Wikifunctions.ZID}\leanok
  A Wikifunctions object id is a string. Models \wf{Z9} references (schema \texttt{Z9.yaml}: \texttt{pattern \^{}Z[1-9]$\backslash$d*\$}).
  Source: \leansrc{Wikifunctions/Core.lean}{16}.
\end{definition}

\begin{definition}[Value universe]\label{def:core-val}
  \lean{Wikifunctions.Val}\leanok
  The subset of \wf{Z4} types the evaluator needs: naturals (\wf{Z13518}), booleans (\wf{Z40}), and rationals (\wf{Z19677}).
  Source: \leansrc{Wikifunctions/Core.lean}{20}.
\end{definition}

\begin{definition}[Composition AST]\label{def:core-expr}
  \uses{def:core-val}
  \lean{Wikifunctions.Expr}\leanok
  Three node shapes: \texttt{arg} (a \wf{Z18} argument reference), \texttt{lit} (an inline literal value), and \texttt{call} (a \wf{Z7} function call naming its callee by \wf{Z9} reference). This is the shape of a \wf{Z14}K2 composite body on the modelled fragment (\divergence{D16}: real \wf{Z18} references are by key-id string, modelled here positionally; \divergence{D17}: general \wf{Z14}K2 bodies admit more forms).
  Source: \leansrc{Wikifunctions/Core.lean}{28}.
\end{definition}

\begin{definition}[Denotation]\label{def:core-denotation}
  \uses{def:core-val}
  \lean{Wikifunctions.Denotation}\leanok
  A function's extensional meaning: \texttt{List Val $\to$ Option Val}, total, with \texttt{none} for a type error or arity mismatch. Models \wf{Z8} extensionally (\wf{Z8}K1 arguments $\to$ \wf{Z8}K2 return), collapsing the \wf{Z22}/\wf{Z5} error envelope to \texttt{Option} (\divergence{D-info}: documented abstraction).
  Source: \leansrc{Wikifunctions/Core.lean}{34}.
\end{definition}

\begin{definition}[Registry]\label{def:core-registry}
  \uses{def:core-denotation, def:core-zid}
  \lean{Wikifunctions.Registry}\leanok
  Resolves a ZID to its denotation --- the global function context, modelling the wiki of \wf{Z2} persistent objects; a missing id yields \texttt{none} (live: \wf{Z504}/\wf{Z550}).
  Source: \leansrc{Wikifunctions/Core.lean}{38}.
\end{definition}

\begin{definition}[Evaluator]\label{def:core-eval}
  \uses{def:core-expr, def:core-registry, def:core-denotation}
  \lean{Wikifunctions.eval, Wikifunctions.evalArgs}\leanok
  A total, mutually-structural evaluator: \texttt{arg} indexes the environment, \texttt{lit} returns its value, \texttt{call} resolves the callee through the registry and applies it to eagerly-evaluated arguments. Matches the deployed orchestrator's eager application on the modelled fragment (the one lazy builtin, \wf{Z802}, is out of scope).
  Source: \leansrc{Wikifunctions/Core.lean}{43}.
\end{definition}

\chapter{The imperative-Python layer}

A deep embedding of the imperative-Python subset used by two \emph{deployed} Wikifunctions code implementations, with a fuel-interpreter semantics, proved equivalent to Mathlib specifications. Trust assumption (\leansrc{Wikifunctions/Python/Imp.lean}{31}): the embedding faithfully models the deployed interpreter on this subset --- deployed user Python runs on RustPython~0.5.0 / wasm (\divergence{D11}).

\section{The embedding (\texttt{Imp.lean})}

\begin{definition}[Variable store]\label{def:imp-state}
  \lean{Wikifunctions.Python.State}\leanok
  A strict association list \texttt{List (String $\times$ Nat)}; first binding wins, unbound reads default to \texttt{0}. Source: \leansrc{Wikifunctions/Python/Imp.lean}{46}.
\end{definition}

\begin{definition}[Store read/write]\label{def:imp-stateops}
  \uses{def:imp-state}
  \lean{Wikifunctions.Python.State.get, Wikifunctions.Python.State.set}\leanok
  \texttt{get} is a single strict linear scan; \texttt{set} conses a shadowing binding (no lazy closure chain --- the fix for the exponential-blowup bug documented in the source). Source: \leansrc{Wikifunctions/Python/Imp.lean}{50}.
\end{definition}

\begin{definition}[Expressions and conditions]\label{def:imp-expr}
  \lean{Wikifunctions.Python.Expr, Wikifunctions.Python.Cond}\leanok
  Expressions: variable, nat literal, \texttt{+}, \texttt{*}, \texttt{\%}. Conditions: \texttt{e != 0} and \texttt{e1 <= e2}. Source: \leansrc{Wikifunctions/Python/Imp.lean}{69}.
\end{definition}

\begin{definition}[Statements]\label{def:imp-stmt}
  \uses{def:imp-expr}
  \lean{Wikifunctions.Python.Stmt}\leanok
  A \emph{parallel} assignment \texttt{x1, x2 = e1, e2} (both right-hand sides evaluated in the pre-state --- a faithful Python swap), a \texttt{while} loop, and sequencing. Source: \leansrc{Wikifunctions/Python/Imp.lean}{85}.
\end{definition}

\begin{definition}[Expression / condition evaluation]\label{def:imp-eval}
  \uses{def:imp-expr, def:imp-stateops}
  \lean{Wikifunctions.Python.Expr.eval, Wikifunctions.Python.Cond.eval}\leanok
  Total evaluation of expressions to \texttt{Nat} and conditions to \texttt{Bool}. \texttt{\%} uses total \texttt{Nat.mod} (\divergence{D5}: Python raises on \texttt{\% 0}; provably unreachable in both deployed programs). Source: \leansrc{Wikifunctions/Python/Imp.lean}{94}.
\end{definition}

\begin{definition}[Parallel assignment semantics]\label{def:imp-dopassign}
  \uses{def:imp-stateops, def:imp-eval}
  \lean{Wikifunctions.Python.doPassign}\leanok
  Both right-hand sides are evaluated in the current state, then both targets updated. Source: \leansrc{Wikifunctions/Python/Imp.lean}{109}.
\end{definition}

\begin{definition}[Fuel interpreter]\label{def:imp-run}
  \uses{def:imp-stmt, def:imp-dopassign, def:imp-eval}
  \lean{Wikifunctions.Python.Stmt.run}\leanok
  Operational semantics of statements; \texttt{none} means the fuel budget was exhausted. Source: \leansrc{Wikifunctions/Python/Imp.lean}{116}.
\end{definition}

\section{\texttt{Z13701} ``are coprime'' --- verified against \texorpdfstring{\texttt{Nat.Coprime}}{Nat.Coprime}}

Models the deployed Python implementation \wfl{Z29182}{impl.\ of \wf{Z13701}}: \texttt{while b != 0: a, b = b, a \% b; return a == 1}.

\begin{definition}[The coprime program]\label{def:z13701-prog}
  \uses{def:imp-stmt, def:imp-state}
  \lean{Wikifunctions.Python.loop, Wikifunctions.Python.initState}\leanok
  A 1:1 transcription of \wf{Z29182} as \texttt{Stmt} data. Source: \leansrc{Wikifunctions/Python/Z13701.lean}{50}. (\divergence{D9}: the proved program and the \texttt{DiffDriver} program are token-identical twins in separate modules.)
\end{definition}

\begin{lemma}[Euclidean invariant]\label{lem:z13701-gcd}
  \uses{def:imp-dopassign}
  \lean{Wikifunctions.Python.gcd_body}\leanok
  The loop body preserves \texttt{Nat.gcd}: after \texttt{a, b = b, a \% b} one has \texttt{gcd b (a \% b) = gcd a b}. Source: \leansrc{Wikifunctions/Python/Z13701.lean}{85}.
  \begin{proof}\leanok Via \texttt{Nat.gcd\_rec} and \texttt{Nat.gcd\_comm}.\end{proof}
\end{lemma}

\begin{lemma}[Loop correctness and termination]\label{lem:z13701-loop}
  \uses{def:imp-run, lem:z13701-gcd, def:z13701-prog}
  \lean{Wikifunctions.Python.loop_correct}\leanok
  With fuel exceeding \texttt{b}, the loop terminates and its final \texttt{a} is \texttt{gcd a b}. Source: \leansrc{Wikifunctions/Python/Z13701.lean}{99}.
  \begin{proof}\leanok Induction on fuel; the recursive step decreases \texttt{b} by \texttt{Nat.mod\_lt}.\end{proof}
\end{lemma}

\begin{definition}[Program result]\label{def:z13701-run}
  \uses{def:z13701-prog, def:imp-run}
  \lean{Wikifunctions.Python.runProgram}\leanok
  Runs the loop with fuel \texttt{b+1}, then returns \texttt{a == 1}. Source: \leansrc{Wikifunctions/Python/Z13701.lean}{136}.
\end{definition}

\begin{theorem}[Coprimality]\label{thm:z13701}
  \uses{def:z13701-run, lem:z13701-loop}
  \lean{Wikifunctions.Python.runProgram_eq_coprime}\leanok
  For all \(a, b\), \texttt{runProgram a b = some (decide (Nat.Coprime a b))} --- the deployed \wf{Z13701} computes exactly Mathlib's \texttt{Nat.Coprime}. Source: \leansrc{Wikifunctions/Python/Z13701.lean}{143}.
  \begin{proof}\leanok From the loop lemma (\ref{lem:z13701-loop}); axioms \texttt{[propext, Classical.choice, Quot.sound]}.\end{proof}
\end{theorem}

\chapter{The composite layer}

A deep embedding of the \wf{Z14}K2 \emph{composition} language with a total evaluator, proving six real composite implementations equal to their Mathlib oracles. Each leaf carries a Mathlib-backed denotation; a composite inherits correctness from its leaves and the evaluator.

\begin{definition}[Composite value universe]\label{def:eval-val}
  \lean{Wikifunctions.Val}\leanok
  Naturals (\wf{Z13518}), booleans (\wf{Z40}), rationals (\wf{Z19677}). (This is \texttt{WikifunctionsEval.lean}'s own \texttt{Val}, distinct from the core value universe (\ref{def:core-val}) --- \divergence{D-info}: the two files define clashing \texttt{Wikifunctions.Val}.) Source: \leansrc{WikifunctionsEval.lean}{58}.
\end{definition}

\begin{definition}[Leaf library]\label{def:eval-leaf}
  \lean{Wikifunctions.Leaf}\leanok

Source: \leansrc{WikifunctionsEval.lean}{66}.

\begin{definition}[Composition AST and evaluator]\label{def:eval-eval}

\paragraph{Community types this corpus rests on.} \wf{Z13518} (natural, a decimal-string record), \wf{Z16683} (integer), \wf{Z19677} (rational: sign/numerator/denominator), \wf{Z20838} (float64).
