% Macros shared by the PDF and web builds.

% Theorem-like environments that become dependency-graph nodes.
\newtheorem{theorem}{Theorem}
\newtheorem{proposition}[theorem]{Proposition}
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{corollary}[theorem]{Corollary}

\theoremstyle{definition}
\newtheorem{definition}[theorem]{Definition}

\theoremstyle{remark}
\newtheorem*{remark}{Remark}

% --- Wikifunctions cross-reference macros (work in both pdf and web) ---
% \wf{Zxxxx} : link to a live Wikifunctions object.
\newcommand{\wf}[1]{\href{https://www.wikifunctions.org/wiki/#1}{\textsf{#1}}}
% \wfl{Zxxxx}{label} : link plus a human label.
\newcommand{\wfl}[2]{\href{https://www.wikifunctions.org/wiki/#1}{\textsf{#1}}~(#2)}
% \leansrc{path}{line} : link to the exact Lean source line on GitHub.
\newcommand{\leansrc}[2]{\href{https://github.com/Deicyde/wikifunctions/blob/main/#1\#L#2}{\texttt{#1:#2}}}
% \fnmodel{anchor}{text} : cite a section of the Abstract Wikipedia Function model
% (the canonical prose spec). Use for the normative statement behind a claim.
\newcommand{\fnmodel}[2]{\href{https://www.wikifunctions.org/wiki/Wikifunctions:Function_model\##1}{#2}}
% \schema{Zxx} : cite the machine-enforced function-schemata JSON-Schema for a type.
\newcommand{\schema}[1]{\href{https://gitlab.wikimedia.org/repos/abstract-wiki/wikifunctions/function-schemata/-/blob/main/data/CANONICAL/#1.yaml}{\texttt{#1.yaml}}}
% \repo{path}{text} : link to a file in this repository.
\newcommand{\repo}[2]{\href{https://github.com/Deicyde/wikifunctions/blob/main/#1}{#2}}
